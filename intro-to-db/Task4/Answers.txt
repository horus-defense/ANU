Question 2
SELECT COUNT(order_item.item_id) AS Items_Count, SUM(unit_price * quantity) AS Total_Price, order_id
FROM order_item JOIN item ON order_item.item_id = item.item_id
GROUP BY order_id;

Output:
Items_Count  Total_Price  order_id
2            1050.00      1
2            1250.00      2
1            600.00       3
1            500.00       4

Question 3
SELECT `order`.order_id FROM `order`
JOIN shipment ON shipment.order_id = `order`.order_id
JOIN warehouse ON warehouse.warehouse_id = shipment.warehouse_id
WHERE warehouse_city = 'Arica'
GROUP BY `order`.order_id;

Output:
order_id
2

Question 4
SELECT SUM(item.unit_price * order_item.quantity) AS Total_Price_Of_Orders
FROM shipment JOIN order_item ON order_item.order_id = shipment.order_id
JOIN item ON order_item.item_id = item.item_id WHERE warehouse_id = 8;

Output:
Total_Price_Of_Orders
1650.00
